import RPi.GPIO as GPIO
import time

# Define GPIO pins for units and tens
units_pins = [4, 17, 18, 27]    # A-D for first 7447
tens_pins = [5, 6, 12, 13]     # A-D for second 7447

GPIO.setmode(GPIO.BCM)
GPIO.setwarnings(False)

# Setup all pins as output
for pin in units_pins + tens_pins:
    GPIO.setup(pin, GPIO.OUT)
    GPIO.output(pin, GPIO.LOW)

def send_bcd(value, pins):
    """Send 4-bit BCD value to specified GPIO pins."""
    for i in range(4):
        GPIO.output(pins[i], (value >> i) & 1)

def main():
   while True:
      for count in range(100):
         tens = count // 10
         units = count % 10
         send_bcd(units, units_pins)
         send_bcd(tens, tens_pins)
         time.sleep(1)
      GPIO.cleanup()

# Execute the main program
if __name__ == '__main__':
   main()
